# =========================================================
# Caddy — HTTPS 종료와 리버스 프록시
#
# 인증서 발급과 갱신을 알아서 한다. H-01 웹 푸시가 HTTPS 없이는
# 동작하지 않으므로, 갱신을 놓쳐 만료되면 시즌 중에 알림이 멈춘다.
#
# DOMAIN 이 비어 있으면 :80 으로 뜬다(도메인 붙기 전 확인용).
# =========================================================

{$DOMAIN::80} {
	encode zstd gzip

	# Next.js 정적 자산은 파일명에 해시가 붙어 있어 오래 캐시해도 안전하다.
	@static path /_next/static/* /favicon.ico /manifest.webmanifest
	header @static Cache-Control "public, max-age=31536000, immutable"

	header {
		X-Content-Type-Options nosniff
		X-Frame-Options DENY
		Referrer-Policy strict-origin-when-cross-origin
		-Server
	}

	reverse_proxy app:3000 {
		# 앱이 요청 출처를 올바르게 판단하려면 이 헤더들이 필요하다.
		header_up X-Real-IP {remote_host}
		health_uri /api/health
		health_interval 30s
	}

	log {
		output file /var/log/caddy/access.log {
			roll_size 20mb
			roll_keep 5
		}
	}
}
